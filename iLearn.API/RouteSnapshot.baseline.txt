# iLearn.API route snapshot — baseline (Phase 0)
#
# Captured manually from `[Route] + [Http*]` attributes because the sandbox
# cannot run `dotnet build` (DevExpress local NuGet feed is Windows-only).
# Use this file as a regression checklist after each refactor phase: any
# additions/removals/renames here represent a public-contract change.
#
# Source: grep -nP '^\s*\[(Route|Http(Get|Post|Put|Delete|Patch))' on every
# controller file at the time of branching from main.
#
# Phase 1–3 refactor (PR): the original `Controllers/Base/CRUDController.cs`
# god file (1,225 LOC, 15 classes) was split into 15 per-contentItem files
# under `Controllers/Base/` (e.g. `CategoriesCRUDController.cs`).
# All routes, controller class names, and HTTP verbs are preserved verbatim.
# `AssignmentsController` and `EnrollmentsController` no longer inject
# `AppDbContext`; the same routes/responses are produced via repositories +
# `IUnitOfWork`.

# --- Base / Generic CRUD (api/admin/[controller]) ---
# api/admin/<Controller> derives the route segment from the C# class name
# minus "Controller" — e.g. CategoriesCRUDController -> api/admin/CategoriesCRUD
GenericController<T>          [Route("api/admin/[controller]")]
  GET    Get
  GET    Get/{id}
  POST   Post
  PUT    Put
  DELETE Delete

# --- Cross-cutting ---
Negotiate (Windows) authentication, FallbackPolicy = DefaultPolicy
Authorization policies: AdminOnly, SuperAdminOnly, ManagerOrAbove, UserOrAbove, DomainUser
CORS policy "AllowSpecificOrigin" registered Development-only,
            but applied unconditionally in pipeline (known bug — Phase 5)
Middleware order: UseCors → UseHttpsRedirection → UseAuthentication
                  → UseMiddleware<ApiClaimsEnrichMiddleware> → UseAuthorization
                  → MapControllers → MapHub
